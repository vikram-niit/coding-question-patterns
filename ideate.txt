1. n-queens problem
The problem is about placing n queens on a nxn chessboard such that no two queens attack each other. The goal is to find as many such solutions as possible.

Approach: Backtracking = recursion + loop

def solve(row, x):
	if row == n:
		print_solution(x)
		return

	for col in range(len(row)):
		# advance the position of the queen in the row
		x[row] += 1
		# check if the position is safe
		if is_safe(row, x):
			solve(row + 1, x)

def is_safe(row, x):
	for i in range(0, row):
		if i + row == x[i] + x[row]:
			return False
		elif abs(i - row]) == abs(x[i] - x[row]):
			return False
		else:
			return True

2. Find the nth highest using heap

Approach: Maintain a min heap containing the top n items
add a new item and pop the item from the heap
After all the items are attempted to be added. Pop the element which would be the nth highest item



import heapq

def nth_highest(x):
	for i in range(len(x)):
		heapq.heappush(x, x[i])
		heapq.pop(x)

	return heapq.pop(x)

def dfs(graph, node, visited):
	# if node is in visited skip and continue
	if node in visited:
		return

	# add node to visited
	visited.add(node)
	result.append(node)

	for child in graph[node]:
		dfs(graph, child, visited)

5. Find max sum subarray

def find_max_sum_sub_array(x, k):
	window_sum = sum(x[0:k])
	max_sum = window_sum
	left = 0
	
	for i in range(k, len(x)):
		window_sum += x[i]
		window_sum -= x[i-k]

		# update max_sum
		if window_sum > max_sum:
			max_sum = window_sum
			left = i

	return max_sum, left

6. Longest substring without repeating characters

def find_longest_substring_Without_repeating_characters(s):
	left = 0
	right = 0
	result = []

	characters = set()
	for i in range(len(s)):
		if s[i] not in characters:
			characters.add(s[i])
			right = i
		elif s[i] in characters:
			result.append((left - right + 1), (left, right))
			left = i
			right += 1
	
	return max(result, key=0)

7. Container with most water

def max_area(heights):
	left = 0
	right = len(heights) - 1
	max_area = 0

	while left <= right:
		current_area = min(heights[left], heights[right]) * (right - left)
		max_area = max(current_area, max_area)
		
		if (heights[left + 1] < heights[right - 1]):
			 left += 1
		else:
			right -= 1

	return max_area

def count_subarray_sum(nums, k):
	current_sum = 0
	prefix_sum = 0
	count = 0

	freq = {0: 1}

	for num in nums:
		prefix_sum += num

		needed = prefix_sum - k

		if needed in freq:
			count += freq[needed]
		freq[prefix_sum] = freq.get(prefix_sum, 0) + 1

12. def binary_search(numbers, key):
	low = 0
	high = len(numbers) - 1
	
	while low < high:
		mid = (low + high) / 2
		if numbers[mid] <= key:
			high = mid - 1
		elif numbers[mid] > key:
			low = mid+ 1
		elif numbers[mid] == key:
			return mid
					
	if low == high and numbers[low] == key:
		return low
	else:
		return -1

13. MST (Minimum Spanning Tree)
Kruskal's algorithm

def Kruskal(edges, vertices):
	# sort the edges by their weights
	# choose n - 1 edges in ascending order

	parent = {v: v for v in vertices}

	def find(v):
		if parent[v] != v:
			parent[v] = find(parent[v])
		return v

	def union(u, v):
		root_u = find(u)
		root_v = find(v)
		if (find(u) == find(v)):
			return False

		parent[root_v] = root_u
		return True
	
	result = []
	edges.sort(key=lambda x: x[1])

	for edge, weight in range(edges):
		u, v = edge
		if union(u, v):
			result.append((edge, weight))
			if len(result) == len(vertices - 1):
				break

	return result


if __name__ == '__main__':
	edges = [(['A', 'B'], 6),
		(['B', 'C'], 7),
		(['D', 'E'], 8),
		(['E', 'A'], 9),
		(['E', 'F'], 10)]

	vertices = ['A', 'B', 'C', 'D', 'E', 'F']

	Kruskal(edges, vertices)
			

14. Prim's algorithm

# start with a vertex
# Then choose the shortest distance neighbour
# Then from the neighbour choose its shortest distance neighbour 
# repeat the process until n - 1 vertices are covered
def prims(graph):
	result = []
	for vertex in graph:
		# select shortest distance neighbour
		neighbours = graph[vertex]
		edge, weight = min(neighbours, key = lambda x: x[1])			
		result.append(vertex, (edge, weight))
	
	return result

import heapq

def dijkstra(graph, start):
    distances = {vertex: float('inf') for vertex in graph}
    distances[start] = 0

    heap = [(0, start)]

    while heap:
        current_distance, current_vertex = heapq.heappop(heap)

        # Ignore outdated heap entries
        if current_distance > distances[current_vertex]:
            continue

        for vertex, weight in graph[current_vertex]:
            new_distance = current_distance + weight

            if new_distance < distances[vertex]:
                distances[vertex] = new_distance
                heapq.heappush(heap, (new_distance, vertex))

    return distances
