def max_sum(arr, k):
    window_sum = sum(arr[:k])
    max_sum = window_sum

    for i in range(k, len(arr)):
        window_sum += arr[i]       # add new element
        window_sum -= arr[i - k]   # remove old element

3. Variable-size window — smallest subarray with sum ≥ target
This is one of the most important sliding-window patterns.

Problem

Solution
def min_subarray(arr, target):
    left = 0
    window_sum = 0
    min_length = float('inf')

    for right in range(len(arr)):
        window_sum += arr[right]

        while window_sum >= target:
            min_length = min(min_length, right - left + 1)

            window_sum -= arr[left]
            left += 1

    return 0 if min_length == float('inf') else min_length

def longest_unique(s):
    left = 0
    chars = set()
    max_length = 0

    for right in range(len(s)):
        while s[right] in chars:
            chars.remove(s[left])
            left += 1

        chars.add(s[right])

        max_length = max(max_length, right - left + 1)

    return max_length

def longest_k_distinct(s, k):
    left = 0
    count = {}
    max_length = 0

    for right in range(len(s)):
        count[s[right]] = count.get(s[right], 0) + 1

        while len(count) > k:
            count[s[left]] -= 1

            if count[s[left]] == 0:
                del count[s[left]]

            left += 1

        max_length = max(max_length, right - left + 1)

    return max_length

from collections import Counter

def count_anagrams(s, pattern):
    k = len(pattern)
    target = Counter(pattern)
    window = Counter()
    result = 0

    for i in range(len(s)):
        window[s[i]] += 1

        if i >= k:
            window[s[i-k]] -= 1

            if window[s[i-k]] == 0:
                del window[s[i-k]]

            result += 1

    return result

def maxArea(height):
    left = 0
    right = len(height) - 1
    max_water = 0

    while left < right:
        width = right - left
        water = min(height[left], height[right]) * width
        max_water = max(max_water, water)

        # Move the pointer with the smaller height
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1

    return max_water

def cycle_start(head):
    slow = fast = head

    # Detect cycle
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next

        if slow == fast:
            break
    else:
        return None

15. def horspool(text, pattern):
    m = len(pattern)
    n = len(text)

    # Create shift table
    shift = {}

    # Default shift for every character is m
    for i in range(m - 1):
        shift[pattern[i]] = m - 1 - i

    # Match pattern against text
    i = m - 1

    while i < n:
        k = 0

        # Compare from right to left
        while k < m and text[i - k] == pattern[m - 1 - k]:
            k += 1

        # Pattern completely matched
        if k == m:
            return True, i - m + 1

        # Shift based on the character aligned with pattern's last character
        i += shift.get(text[i], m)

    return False, -1

    pattern = "mice"

    found, index = horspool(text, pattern)

    if found:
        print(f"Pattern found at index {index}")
    else:
        print("Pattern not found")

That's the main idea behind Horspool: use the mismatching/aligned text character to skip possible matches.

The key Horspool idea
For:

text:    The cat chased the mice
pattern:                  mice
                         ↑
                    compare from here

Horspool compares the pattern right-to-left.

If there is a mismatch, it doesn't simply move one position. It uses the shift table to skip positions that cannot possibly match.

shift.get(text[i], m)

The character text[i] is the character currently aligned with the last character of the pattern, and its shift determines how far we jump.

14. def valid_parentheses(expression):

15. def binary_search(numbers, key):
    low = 0
    high = len(numbers) - 1

    while low <= high:
        mid = (low + high) // 2

        if numbers[mid] == key:
            return mid
        elif numbers[mid] < key:
            low = mid + 1
        else:
            high = mid - 1

    return -1

16. def Kruskal(edges, vertices):
    # Sort edges by weight
    edges.sort(key=lambda x: x[1])

    parent = {v: v for v in vertices}

    def find(v):
        if parent[v] != v:
            parent[v] = find(parent[v])
        return parent[v]

    def union(u, v):
        root_u = find(u)
        root_v = find(v)

        if root_u == root_v:
            return False

        parent[root_v] = root_u
        return True

    result = []

    for edge, weight in edges:
        u, v = edge

        if union(u, v):
            result.append((edge, weight))

            if len(result) == len(vertices) - 1:
                break

    return result

    edges = [
        (['A', 'B'], 6),
        (['B', 'C'], 7),
        (['D', 'E'], 8),
        (['E', 'A'], 9),
        (['E', 'F'], 10)
    ]

    vertices = ['A', 'B', 'C', 'D', 'E', 'F']

    print(Kruskal(edges, vertices))

17. Prim's algorithm

import heapq

def prim(graph, start=0):
    visited = set()
    heap = [(0, start, -1)]
    mst = []
    total = 0

    while heap:
        weight, vertex, parent = heapq.heappop(heap)

        if vertex in visited:
            continue

        visited.add(vertex)
        total += weight

        if parent != -1:
            mst.append((parent, vertex, weight))

        for neighbour, weight in graph[vertex]:
            if neighbour not in visited:
                heapq.heappush(heap, (weight, neighbour, vertex))

    return mst, total

    	4: [(1, 5), (2, 7), (3, 9)]
	}

18. Dijkstra's algorithm

import heapq

def dijkstra(graph, start):
    distances = {vertex: float('inf') for vertex in graph}
    distances[start] = 0

    heap = [(0, start)]

    while heap:
        current_distance, current_vertex = heapq.heappop(heap)

        # Ignore outdated heap entries
        if current_distance > distances[current_vertex]:
            continue

        for vertex, weight in graph[current_vertex]:
            new_distance = current_distance + weight

            if new_distance < distances[vertex]:
                distances[vertex] = new_distance
                heapq.heappush(heap, (new_distance, vertex))

    return distances


if __name__ == '__main__':
    graph = {
        'A': [('B', 5), ('C', 10)],
        'B': [('C', 3)],
        'C': []
    }

    print(dijkstra(graph, 'A'))

from fastapi import FastAPI, HTTPException

app = FastAPI()


class StandardShipping:
    def calculate(self, order):
        return 100


class ExpressShipping:
    def calculate(self, order):
        return 300


class InternationalShipping:
    def calculate(self, order):
        return 1000


class ShippingFactory:

    strategies = {
        "standard": StandardShipping,
        "express": ExpressShipping,
        "international": InternationalShipping,
    }

    @classmethod
    def create(cls, shipping_type):
        strategy = cls.strategies.get(shipping_type)

        if not strategy:
            raise ValueError("Invalid shipping type")

        return strategy()

Now your API becomes:

        strategy = ShippingFactory.create(shipping)
    except ValueError:
        raise HTTPException(
            status_code=400,
            detail="Invalid shipping type"
        )

    order = Order(amount, strategy)

    return {
        "total": order.calculate_total()
    }

Now you've moved the decision from the controller to the factory.

But notice something important:

    "standard": StandardShipping,
    "express": ExpressShipping,
    "international": InternationalShipping,
}

You still have to modify the factory when adding a new shipping type.

For example:

    def calculate(self, order):

    "standard": StandardShipping,
    "express": ExpressShipping,
    "international": InternationalShipping,

That's usually acceptable. The goal isn't "never change any code."

The goal is:

Adding a new behavior should require adding new code, not modifying existing business logic in many places.

An even cleaner approach: registration
You can make the strategies self-register:

class ShippingFactory:

    strategies = {}

    @classmethod
    def register(cls, name, strategy):
        cls.strategies[name] = strategy

    @classmethod
    def create(cls, name):
        if name not in cls.strategies:
            raise ValueError("Invalid shipping type")

        return cls.strategies[name]()

        return 500

    def pay(self, amount):
        self.stripe.make_payment(amount)

Problem solved: Making incompatible interfaces work together.

33. Template Method

4. Repository Pattern
The application shouldn't care whether orders are stored in PostgreSQL, MongoDB, Redis, etc.

Define a repository:

# repositories/order_repository.py

class OrderRepository:

    def __init__(self):
        self.orders = {}

    def save(self, order):
        self.orders[order.id] = order

    def get(self, order_id):
        return self.orders.get(order_id)

Our service uses:

repository.save(order)

rather than:

postgres.execute(...)

This separates:

Business logic
      |
      v
Repository
      |
      v

PostgresOrderRepository

5. Strategy Pattern
Shipping is a perfect Strategy example.

# shipping/strategy.py

from abc import ABC, abstractmethod


class ShippingStrategy(ABC):

    @abstractmethod
    def calculate(self, amount):
        pass

Standard:

# shipping/standard.py

from .strategy import ShippingStrategy


class StandardShipping(ShippingStrategy):

    def calculate(self, amount):
        return 100

Express:

# shipping/express.py

from .strategy import ShippingStrategy


class ExpressShipping(ShippingStrategy):

    def calculate(self, amount):
        return 300

International:

# shipping/international.py

from .strategy import ShippingStrategy


class InternationalShipping(ShippingStrategy):

    def calculate(self, amount):
        return 1000

Now the application doesn't need:

if shipping == "standard":
    ...
elif shipping == "express":
    ...

inside the business logic.

6. Factory Pattern
We need something that selects the correct strategy.

# shipping/factory.py

from .standard import StandardShipping
from .express import ExpressShipping
from .international import InternationalShipping


class ShippingFactory:

    strategies = {
        "standard": StandardShipping,
        "express": ExpressShipping,
        "international": InternationalShipping,
    }

    @classmethod
    def create(cls, shipping_type):

        strategy_class = cls.strategies.get(shipping_type)

        if not strategy_class:
            raise ValueError("Invalid shipping type")

        return strategy_class()

Notice again:

"express": ExpressShipping

is the class.

The factory creates the object:

strategy_class()

So:

"express"
    |
    v
ExpressShipping     <- class
    |
    | ()
    v
ExpressShipping()   <- object

7. Adapter Pattern
Suppose our application expects:

payment.pay(amount)

But an external payment provider gives us:

stripe.charge(amount)

We don't want our entire application to depend on Stripe's API.

Create an adapter:

# payments/stripe_adapter.py

class StripeClient:

    def charge(self, amount):
        print(f"Stripe charged {amount}")


class StripePaymentAdapter:

    def __init__(self, stripe_client):
        self.stripe_client = stripe_client

    def pay(self, amount):
        self.stripe_client.charge(amount)

Now our application knows only:

payment.pay(5000)

It doesn't care whether the underlying provider is:

Stripe
PayPal
Razorpay
Adyen

The Adapter translates our interface into the external provider's interface.

8. Service Layer
Now we put business logic into a service.

# services/order_service.py

class OrderService:

    def __init__(
        self,
        repository,
        shipping_factory,
        payment
    ):
        self.repository = repository
        self.shipping_factory = shipping_factory
        self.payment = payment

    def create_order(
        self,
        order,
        shipping_type
    ):

        shipping = self.shipping_factory.create(
            shipping_type
        )

        shipping_cost = shipping.calculate(
            order.amount
        )

        total = order.amount + shipping_cost

        self.payment.pay(total)

        self.repository.save(order)

        return {
            "order_id": order.id,
            "amount": order.amount,
            "shipping": shipping_cost,
            "total": total
        }

Look carefully at the constructor:

def __init__(
    self,
    repository,
    shipping_factory,
    payment
):

This is Dependency Injection.

OrderService does NOT do:

self.repository = OrderRepository()

or:

self.payment = StripePaymentAdapter(...)

Those dependencies are supplied from outside.

9. Dependency Injection with FastAPI
Now FastAPI can construct the dependencies.

from fastapi import Depends


def get_order_repository():
    return OrderRepository()


def get_payment():

    stripe = StripeClient()

    return StripePaymentAdapter(stripe)


def get_order_service(
    repository=Depends(get_order_repository),
    payment=Depends(get_payment)
):

    return OrderService(
        repository=repository,
        shipping_factory=ShippingFactory,
        payment=payment
    )

Then:

@app.post("/orders")
def create_order(
    amount: float,
    shipping: str,
    service: OrderService = Depends(get_order_service)
):

    order = Order(
        order_id="123",
        amount=amount
    )

    return service.create_order(
        order,
        shipping
    )

FastAPI effectively performs:

FastAPI
   |
   +--> OrderRepository()
   |
   +--> StripeClient()
   |        |
   |        v
   |   StripePaymentAdapter
   |
   +--> OrderService(...)
            |
            v
       Controller

That's Dependency Injection in practice.

10. Facade Pattern
Our order process might eventually become complicated:

Validate order
    ↓
Calculate shipping
    ↓
Charge payment
    ↓
Save order
    ↓
Publish event
    ↓
Send notification

Rather than making the controller know all of this, create a Facade:

class OrderFacade:

    def __init__(
        self,
        order_service,
        event_publisher
    ):
        self.order_service = order_service
        self.event_publisher = event_publisher

    def place_order(
        self,
        order,
        shipping_type
    ):

        result = self.order_service.create_order(
            order,
            shipping_type
        )

        self.event_publisher.publish(
            "order_created",
            order
        )

        return result

Now the controller only needs:

facade.place_order(order, shipping)

It doesn't need to understand the entire workflow.

11. Observer Pattern
When an order is created, several things might need to happen:

Order Created
     |
     +----> Email
     |
     +----> Analytics
     |
     +----> Inventory

We can create an event publisher:

class EventPublisher:

    def __init__(self):
        self.listeners = {}

    def subscribe(self, event, listener):

        self.listeners.setdefault(
            event,
            []
        ).append(listener)

    def publish(self, event, data):

        for listener in self.listeners.get(event, []):
            listener(data)

Listeners:

def send_email(order):
    print(f"Email sent for order {order.id}")


def update_analytics(order):
    print(f"Analytics updated for {order.id}")

Register:

publisher = EventPublisher()

publisher.subscribe(
    "order_created",
    send_email
)

publisher.subscribe(
    "order_created",
    update_analytics
)

When:

publisher.publish("order_created", order)

both listeners run.

This is the Observer Pattern.

12. State Pattern
Orders have states:

Pending
   |
   v

   |
   v
Delivered

Instead of having massive logic:

    ...
elif status == "shipped":
    ...

    ...

we can represent states as objects.

class PendingState:

    def ship(self, order):
        order.state = ShippedState()

    def deliver(self, order):
        order.state = DeliveredState()


class DeliveredState:
    pass

Then:

The object's behavior changes based on its current state.

13. Command Pattern

        self.order = order
        self.shipping_type = shipping_type

14. Chain of Responsibility
Suppose before creating an order we need:

Authentication
       ↓
Authorization
       ↓
Validation
       ↓
Create Order

Each handler can pass the request to the next handler.

class Handler:

    def __init__(self, next_handler=None):
        self.next = next_handler

    def handle(self, request):

        if self.next:
            return self.next.handle(request)

Authentication:

class AuthenticationHandler(Handler):

    def handle(self, request):

        if not request.authenticated:
            raise Exception("Not authenticated")

        return super().handle(request)

Validation:

class ValidationHandler(Handler):

    def handle(self, request):

        if request.amount <= 0:
            raise Exception("Invalid amount")

        return super().handle(request)

Build the chain:

chain = AuthenticationHandler(
    ValidationHandler()
)

Then:

chain.handle(request)

Flow:

Request
   |
   v
Authentication
   |
   v
Validation
   |
   v
Next handler

This is Chain of Responsibility.

17. What each pattern is doing
The easiest way to remember this isn't by memorizing definitions.

Ask "What problem am I trying to solve?"

Problem	Pattern
Different shipping algorithms	Strategy
Need to create the right strategy	Factory
Don't want service creating dependencies	Dependency Injection
Don't want business logic coupled to DB	Repository
Keep business operations together	Service Layer
External API has a different interface	Adapter
Complex workflow needs a simple entry point	Facade
Add logging/caching without modifying service	Decorator
Notify multiple components about an event	Observer
Object behaves differently by status	State
Represent an operation as an object	Command
Process request through multiple handlers	Chain of Responsibility

One very important lesson
You shouldn't implement all these patterns in every application.

For example, if your application only has:

Order
   ↓
Database

you probably don't need:

Facade
Factory
Strategy
Adapter
Decorator
Observer
Command
State
Chain
...

That would actually make the application more complicated.

Instead, start with simple code:

API
 ↓
Service
 ↓
Repository
 ↓
Database

Then introduce a pattern when you encounter a real problem:

Need multiple algorithms?
        ↓
     Strategy

Need different implementations?
        ↓
      Factory

Need external API integration?
        ↓
      Adapter

Need multiple subscribers?
        ↓
      Observer

Need state-dependent behavior?
        ↓
       State

That's how design patterns are normally useful in real software: they're solutions to recurring design problems, not requirements that every application must satisfy.

38. Toy self-attention example
For "I love cats", suppose the embeddings are:

Create Q, K, V by multiplying each vector by learned matrices:

Q = XWQ,  K = XWK,  V = XWV

Compare queries with keys:

QKᵀ

This tells each token which other tokens are relevant.

Apply softmax to get attention weights:

"cats" → [0.25, 0.25, 0.50]

Meaning it gets information roughly 25% from "I", 25% from "love", and 50% from "cats".

Combine the V vectors using those weights:

attention output = weights × V

So the original token vectors become new context-aware vectors.

In one line:

Q asks what I'm looking for, K tells what each token offers, and V provides the information; attention uses Q–K similarity to decide how much V to take from each token.
